#!/usr/bin/env python3

import os
import time
import random
from datetime import datetime

import requests
from colorama import Fore, Style, init

init(autoreset=True)

BOT_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN")
CHAT_ID = os.getenv("TELEGRAM_CHAT_ID")
INTERVAL = 10


def send_telegram(message):
    if not BOT_TOKEN or not CHAT_ID:
        print(
            Fore.YELLOW +
            "[!] Configura TELEGRAM_BOT_TOKEN y TELEGRAM_CHAT_ID."
        )
        return False

    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"

    try:
        response = requests.post(
            url,
            data={"chat_id": CHAT_ID, "text": message},
            timeout=10
        )

        response.raise_for_status()

        return response.json().get("ok", False)

    except requests.RequestException as error:
        print(Fore.RED + f"[!] Error con Telegram: {error}")
        return False


def generar_evento():
    eventos = [
        ("INFO", "Acceso correcto"),
        ("INFO", "Actividad normal"),
        ("WARNING", "Actividad inusual"),
        ("WARNING", "Intento de acceso fallido"),
        ("CRITICAL", "Actividad sospechosa")
    ]

    nivel, descripcion = random.choice(eventos)
    fecha = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    return nivel, descripcion, fecha


def main():
    print("=" * 50)
    print("          Telegrama-Monitor")
    print("=" * 50)
    print("[+] Monitor iniciado.")
    print("[+] Modo: simulación")

    while True:
        try:
            nivel, evento, fecha = generar_evento()

            mensaje = (
                "🔎 Telegrama-Monitor\n\n"
                f"📅 Fecha: {fecha}\n"
                f"📊 Nivel: {nivel}\n"
                f"📝 Evento: {evento}"
            )

            print(mensaje)

            send_telegram(mensaje)

            time.sleep(INTERVAL)

        except KeyboardInterrupt:
            print("\n[+] Monitor detenido.")
            break

        except Exception as error:
            print(Fore.RED + f"[!] Error: {error}")
            time.sleep(INTERVAL)


if __name__ == "__main__":
    main()

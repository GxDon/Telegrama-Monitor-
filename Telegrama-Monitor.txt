#!/usr/bin/env python3

import os
import time
import random
from datetime import datetime

import requests
from colorama import Fore, Style, init


init(autoreset=True)

BOT_TOKEN = os.getenv("TELEGRAM_BOT_TOKEN")
CHAT_ID = os.getenv("TELEGRAM_CHAT_ID")

INTERVAL = 10


def send_telegram(message):
    if not BOT_TOKEN or not CHAT_ID:
        print(
            Fore.YELLOW
            + "[!] TELEGRAM_BOT_TOKEN o TELEGRAM_CHAT_ID no están configurados."
        )
        return False

    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"

    data = {
        "chat_id": CHAT_ID,
        "text": message
    }

    try:
        response = requests.post(
            url,
            data=data,
            timeout=10
        )

        response.raise_for_status()

        result = response.json()

        if result.get("ok"):
            return True

        print(Fore.RED + "[!] Telegram rechazó el mensaje.")
        return False

    except requests.RequestException as error:
        print(Fore.RED + f"[!] Error de conexión con Telegram: {error}")
        return False


def generar_evento():
    eventos = [
        ("INFO", "Acceso correcto"),
        ("INFO", "Actividad normal"),
        ("WARNING", "Actividad inusual"),
        ("WARNING", "Intento de acceso fallido"),
        ("CRITICAL", "Actividad sospechosa")
    ]

    nivel, descripcion = random.choice(eventos)

    fecha = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    return nivel, descripcion, fecha


def main():
    print("=" * 55)
    print("             Telegrama-Monitor")
    print("=" * 55)

    if not BOT_TOKEN:
        print(Fore.YELLOW + "[!] Falta TELEGRAM_BOT_TOKEN.")

    if not CHAT_ID:
        print(Fore.YELLOW + "[!] Falta TELEGRAM_CHAT_ID.")

    print("[+] Monitor iniciado.")
    print("[+] Modo: simulación")
    print(f"[+] Intervalo: {INTERVAL} segundos")
    print()

    while True:
        try:
            nivel, evento, fecha = generar_evento()

            mensaje = (
                "🔎 Telegrama-Monitor\n\n"
                f"📅 Fecha: {fecha}\n"
                f"📊 Nivel: {nivel}\n"
                f"📝 Evento: {evento}"
            )

            if nivel == "CRITICAL":
                print(Fore.RED + f"[!] {mensaje}")

            elif nivel == "WARNING":
                print(Fore.YELLOW + f"[!] {mensaje}")

            else:
                print(Fore.GREEN + f"[+] {mensaje}")

            send_telegram(mensaje)

            time.sleep(INTERVAL)

        except KeyboardInterrupt:
            print("\n[+] Monitor detenido.")
            break

        except Exception as error:
            print(Fore.RED + f"[!] Error inesperado: {error}")
            time.sleep(INTERVAL)


if __name__ == "__main__":
    main()
